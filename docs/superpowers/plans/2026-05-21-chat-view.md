```typescript
  }

  if (type === 'user_message') {

      <div className="flex justify-end px-3">
        <div className="group relative bg-warm-brown text-white px-3 py-2 rounded-xl rounded-br-sm max-w-[75%] text-[13px] leading-relaxed">
          {message.quoteMessageId && (
            <div className="text-[10px] opacity-60 mb-1 border-l-2 border-white/30 pl-2">
              {content.startsWith('↩ ') ? content.split('\n')[0] : '↩ 引用回复'}
            </div>

          <div className="whitespace-pre-wrap break-words">

          </div>

            title="引用回复"
          >
            <Icon name="corner-down-left" size={12} className="text-warm-text-secondary" />
          </button>
        </div>
      </div>
    );
  }

  // AI bubble (execution, decision, or review)
  const isDecision = type === 'ai_decision';
  const isReview = type === 'ai_review';

  let borderStyle = 'border-warm-border';
  let statusBadge: string | null = null;
  if (type === 'ai_execution') {
    borderStyle = 'border-warm-border';
    statusBadge = '⚡ 执行中';
  } else if (isDecision) {
    borderStyle = 'border-warm-brown';

  } else if (isReview) {
    borderStyle = 'border-green-300 bg-green-50';
    statusBadge = null;
  }

  return (
    <div className="flex justify-start px-3">
      <div className={`group relative bg-white border ${borderStyle} px-3 py-2.5 rounded-xl rounded-bl-sm max-w-[75%] text-[13px] leading-relaxed shadow-sm`}>

              {statusBadge}
            </span>

        {(type === 'ai_execution' && content) && (

                  </button>
                ))}

                  className="px-3 py-1.5 border border-dashed border-warm-border rounded-md text-[11px] text-warm-text-secondary hover:bg-gray-50 transition-colors"
                >
                  ✍️ 自定义
                </button>
              </div>
            )}
          </>

        {/* Review content */}
        {isReview && (
          <>
            <div className="mb-2 whitespace-pre-wrap text-green-900">{content}</div>
            <div className="flex gap-2">

                onClick={() => onApprove(message)}
                className="px-4 py-1.5 bg-green-500 text-white rounded-md text-[11px] hover:bg-green-600 transition-colors"

          title="引用回复"
        >

        </button>

    </div>
  );
```

```typescript
import { useState, useRef, useEffect, type KeyboardEvent } from 'react';
import type { Project } from '../../../src/types';
import type { ChatMessage } from '../App';
import { Icon } from './Icon';

interface Props {
  projects: { id: string; name: string }[];
  quoteTarget: ChatMessage | null;
  onCancelQuote: () => void;
  onSend: (text: string, quoteMessageId?: string) => void;
}

export function ChatInput({ projects, quoteTarget, onCancelQuote, onSend }: Props) {
  const [text, setText] = useState('');
  const [showMention, setShowMention] = useState(false);
  const [mentionFilter, setMentionFilter] = useState('');
  const [mentionIndex, setMentionIndex] = useState(0);
  const inputRef = useRef<HTMLInputElement>(null);

  const filteredProjects = projects.filter(p =>
    p.name.toLowerCase().includes(mentionFilter.toLowerCase())
  );

  useEffect(() => {
    // Detect @mention pattern
    const atMatch = text.match(/@(\S*)$/);
    if (atMatch) {
      setShowMention(true);
      setMentionFilter(atMatch[1]);
      setMentionIndex(0);
    } else {
      setShowMention(false);
    }
  }, [text]);

  const insertMention = (projectName: string) => {
    const newText = text.replace(/@\S*$/, `@${projectName} `);
    setText(newText);
    setShowMention(false);
    inputRef.current?.focus();
  };

  const handleKeyDown = (e: KeyboardEvent) => {
    if (showMention && filteredProjects.length > 0) {
      if (e.key === 'ArrowDown') {
        e.preventDefault();
        setMentionIndex(i => Math.min(i + 1, filteredProjects.length - 1));
      } else if (e.key === 'ArrowUp') {
        e.preventDefault();
        setMentionIndex(i => Math.max(i - 1, 0));
      } else if (e.key === 'Enter' || e.key === 'Tab') {
        e.preventDefault();
        insertMention(filteredProjects[mentionIndex].name);
      } else if (e.key === 'Escape') {
        setShowMention(false);
      }
      return;
    }

    if (e.key === 'Enter' && !e.shiftKey && text.trim()) {
      e.preventDefault();
      onSend(text.trim(), quoteTarget?.id);
      setText('');
    }
  };

  return (
    <div className="border-t border-warm-border bg-white px-3 py-2.5">
      {/* Quote preview */}
      {quoteTarget && (
        <div className="flex items-center gap-2 mb-2 px-2 py-1.5 bg-gray-50 rounded-md text-[11px] text-warm-text-secondary">
          <Icon name="corner-down-left" size={12} />
          <span className="flex-1 truncate">
            引用：{quoteTarget.content.slice(0, 60)}{quoteTarget.content.length > 60 ? '...' : ''}
          </span>
          <button onClick={onCancelQuote} className="hover:text-warm-brown">
            <Icon name="x" size={12} />
          </button>
        </div>
      )}

      {/* Mention dropdown */}
      {showMention && filteredProjects.length > 0 && (
        <div className="mb-2 border border-warm-border rounded-md bg-white shadow-sm max-h-32 overflow-y-auto">
          {filteredProjects.map((p, i) => (
            <button
              key={p.id}
              className={`w-full text-left px-3 py-1.5 text-[12px] hover:bg-gray-50 flex items-center gap-2 ${
                i === mentionIndex ? 'bg-gray-50' : ''
              }`}
              onClick={() => insertMention(p.name)}
            >
              <span className="w-1.5 h-1.5 rounded-full bg-warm-brown/50" />
              {p.name}
            </button>
          ))}
        </div>
      )}

      {/* Input row */}
      <div className="flex items-center gap-2">
        <input
          ref={inputRef}
          type="text"
          value={text}
          onChange={e => setText(e.target.value)}
          onKeyDown={handleKeyDown}
          placeholder="@项目名 描述你要做的事情..."
          className="flex-1 px-3 py-2 border border-warm-border rounded-lg text-[13px] outline-none focus:border-warm-brown transition-colors placeholder:text-gray-300"
        />
        <button
          onClick={() => {
            if (text.trim()) {
              onSend(text.trim(), quoteTarget?.id);
              setText('');
            }
          }}
          disabled={!text.trim()}
          className="p-2 rounded-lg bg-warm-brown text-white disabled:opacity-40 hover:bg-warm-brown/90 transition-colors"
        >
          <Icon name="send" size={18} />
        </button>
      </div>

      <div className="text-[10px] text-warm-text-secondary mt-1.5 px-1">
        💡 @项目名 创建新任务 | 右键/悬停消息引用回复 | 直接点击 AI 消息中的按钮做决策
      </div>
    </div>
  );
}
```

```typescript
import { useRef, useEffect } from 'react';
import type { ChatMessage } from '../App';
import { ChatBubble } from './ChatBubble';

interface Props {
  messages: ChatMessage[];
  onDecisionClick: (msg: ChatMessage, answer: string) => void;
  onApprove: (msg: ChatMessage) => void;
  onReject: (msg: ChatMessage) => void;
  onQuoteReply: (msg: ChatMessage) => void;
}

export function ChatMessageList({ messages, onDecisionClick, onApprove, onReject, onQuoteReply }: Props) {
  const bottomRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    bottomRef.current?.scrollIntoView({ behavior: 'smooth' });
  }, [messages.length, messages[messages.length - 1]?.content.length]);

  if (messages.length === 0) {
    return (
      <div className="flex-1 flex items-center justify-center">
        <div className="text-center text-warm-text-secondary">
          <div className="text-3xl mb-3">💬</div>
          <div className="text-[13px] font-medium mb-1">欢迎来到群聊</div>
          <div className="text-[11px]">

          </div>
        </div>
      </div>
    );

  return (
    <div className="flex-1 overflow-y-auto py-3 flex flex-col gap-2 bg-gray-50/50">
      {messages.map(msg => (
        <ChatBubble
          key={msg.id}
          message={msg}
          onDecisionClick={onDecisionClick}
          onApprove={onApprove}
          onReject={onReject}
```

```typescript
// === Chat: handle WS messages for chat view ===
const setChatMessagesRef = useRef(setChatMessages);
setChatMessagesRef.current = setChatMessages;

// Track task status transitions for chat bubble lifecycle
if (msg.type === 'task_updated') {
  const task = msg.payload as Task;
  const prevStatus = lastTaskStatus.current[task.id];
  lastTaskStatus.current[task.id] = task.status;

  const project = projects.find(p => p.id === task.project_id);
  const projectName = project?.name || task.project_id;

  if (task.status === 'running' && prevStatus && prevStatus !== 'pending') {
    // Resumed after deciding or rejected — new AI execution bubble
    const round = (taskRounds.current[task.id] || 0) + 1;
    taskRounds.current[task.id] = round;
    setChatMessagesRef.current(prev => [...prev, {
      id: chatMsgId(),
      taskId: task.id,
      projectId: task.project_id,
      projectName,
      type: 'ai_execution',
      content: '',
      round,
      timestamp: new Date().toISOString(),
    }]);
  } else if (task.status === 'running' && (!prevStatus || prevStatus === 'pending')) {
    // First execution
    taskRounds.current[task.id] = 0;
    setChatMessagesRef.current(prev => [...prev, {
      id: chatMsgId(),
      taskId: task.id,
      projectId: task.project_id,
      projectName,
      type: 'ai_execution',
      content: '',
      round: 0,
      timestamp: new Date().toISOString(),
    }]);
  }

  if (task.status === 'deciding') {
    // Freeze latest ai_execution bubble, convert to ai_decision
    setChatMessagesRef.current(prev => {
      const updated = [...prev];
      for (let i = updated.length - 1; i >= 0; i--) {
        if (updated[i].taskId === task.id && updated[i].type === 'ai_execution') {
          updated[i] = { ...updated[i], type: 'ai_decision' as const };
          break;
        }
      }
      return updated;
    });
  }

  if (task.status === 'reviewing') {
    // Freeze latest ai_execution, convert to ai_review with summary
    setChatMessagesRef.current(prev => {
      const updated = [...prev];
      for (let i = updated.length - 1; i >= 0; i--) {
        if (updated[i].taskId === task.id && updated[i].type === 'ai_execution') {
          updated[i] = {
            ...updated[i],
            type: 'ai_review' as const,
            content: task.summary || updated[i].content || '任务已完成',
          };
          break;
        }
      }
      return updated;
    });
  }

  if (task.status === 'done') {
    // System message
    setChatMessagesRef.current(prev => [...prev, {
      id: chatMsgId(),
      taskId: task.id,
      projectId: task.project_id,
      projectName,
      type: 'system',
      content: `✅ ${projectName} 任务完成`,
      round: taskRounds.current[task.id] || 0,
      timestamp: new Date().toISOString(),
    }]);
  }
}

if (msg.type === 'task_created') {
  const task = msg.payload as Task;
  lastTaskStatus.current[task.id] = 'pending';
}

if (msg.type === 'decision_created') {
  const { task, decision } = msg.payload as { task: Task; decision: Decision };
  lastTaskStatus.current[task.id] = 'deciding';
  // Attach decision data to the ai_decision bubble
  setChatMessagesRef.current(prev => {
    const updated = [...prev];
    for (let i = updated.length - 1; i >= 0; i--) {
      if (updated[i].taskId === task.id && updated[i].type === 'ai_decision') {
        updated[i] = {
          ...updated[i],
          content: decision.question,
          decision,
        };
        break;
      }
    }
    return updated;
  });
}

if (msg.type === 'task_output') {
  const payload = msg.payload as TaskOutputPayload;
  // Append output to latest ai_execution bubble for this task
  setChatMessagesRef.current(prev => {
    const updated = [...prev];
    for (let i = updated.length - 1; i >= 0; i--) {
      if (updated[i].taskId === payload.taskId && updated[i].type === 'ai_execution') {
        updated[i] = {
          ...updated[i],
          content: updated[i].content + payload.text,
        };
        break;
      }
    }
    return updated;
  });
}
```

```typescript
// === Chat: user action handlers ===
const handleChatSend = useCallback(async (text: string, quoteMessageId?: string) => {
  // Parse @ProjectName from text
  const atMatch = text.match(/^@(\S+)\s+(.+)/);
  if (!atMatch) return; // invalid: no @mention

  const [, projectName, description] = atMatch;
  const project = projects.find(p => p.name === projectName);
  if (!project) return; // project not found

  // Create task
  const task = await createTask(project.id, { title: description.slice(0, 80), description });
  // Start immediately
  await startTask(task.id);

  // Add user message to chat
  const userMsg: ChatMessage = {
    id: chatMsgId(),
    taskId: task.id,
    projectId: project.id,
    projectName: project.name,
    type: 'user_message',
    content: text,
    quoteMessageId,
    round: 0,
    timestamp: new Date().toISOString(),
  };
  setChatMessages(prev => [...prev, userMsg]);

  // Track initial status
  lastTaskStatus.current[task.id] = 'pending';

  // Refresh data
  loadProjects();
  loadTasks(project.id);
}, [projects, loadProjects, loadTasks]);

const handleChatDecision = useCallback(async (msg: ChatMessage, answer: string) => {
  let finalAnswer = answer;
  if (answer === '__custom__') {
    finalAnswer = prompt('输入你的回答:') || '';
    if (!finalAnswer) return;
  }

  // Add user reply message with quote
  const quoteContent = msg.content.slice(0, 100);
  const replyMsg: ChatMessage = {
    id: chatMsgId(),
    taskId: msg.taskId,
    projectId: msg.projectId,
    projectName: msg.projectName,
    type: 'user_message',
    content: `↩ ${quoteContent}\n${finalAnswer}`,
    quoteMessageId: msg.id,
    round: msg.round,
    timestamp: new Date().toISOString(),
  };
  setChatMessages(prev => [...prev, replyMsg]);

  // Submit decision
  await submitDecision(msg.taskId, finalAnswer);
  handleRefresh(msg.projectId);
}, [handleRefresh]);

const handleChatApprove = useCallback(async (msg: ChatMessage) => {
  await approveTask(msg.taskId);
  setChatMessages(prev => [...prev, {
    id: chatMsgId(),
    taskId: msg.taskId,
    projectId: msg.projectId,
    projectName: msg.projectName,
    type: 'system',
    content: `✅ ${msg.projectName} 任务已批准完成`,
    round: msg.round,
    timestamp: new Date().toISOString(),
  }]);
  handleRefresh(msg.projectId);
}, [handleRefresh]);

const handleChatReject = useCallback(async (msg: ChatMessage) => {
  const feedback = prompt('输入驳回原因（可选，留空直接驳回）:') || '';
  await rejectTask(msg.taskId, feedback);
  // User message for the rejection
  const rejectMsg: ChatMessage = {
    id: chatMsgId(),
    taskId: msg.taskId,
    projectId: msg.projectId,
    projectName: msg.projectName,
    type: 'user_message',
    content: feedback ? `↩ 驳回：${feedback}` : '↩ 驳回，请重新修改',
    quoteMessageId: msg.id,
    round: msg.round,
    timestamp: new Date().toISOString(),
  };
  setChatMessages(prev => [...prev, rejectMsg]);
  handleRefresh(msg.projectId);
}, [handleRefresh]);
```
